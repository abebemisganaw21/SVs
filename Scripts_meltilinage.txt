# 3. EXTRACT HAPLOTYPE INFORMATION
# ============================================================================
#### Note CA1 to WA1 is accession name cod by Authors
samples <- c("CA1", "EA5", "WA2", "GA1", "LA1", "LA2", "PA1", "EA3", "WA1")

split_phased_genotype <- function(gt) {
  if(gt %in% c("./.", "0/0", "1/1", "0/1", "1/0")) {
    parts <- strsplit(gt, "/")[[1]]
    return(list(hap1 = as.numeric(parts[1]), hap2 = as.numeric(parts[2])))
  } else {
    return(list(hap1 = NA_real_, hap2 = NA_real_))
  }
}

cat("Extracting haplotype information...\n")
for(sample in samples) {
  haps <- lapply(sv_data[[sample]], function(x) split_phased_genotype(x))
  sv_data[[paste0(sample, "_hap1")]] <- sapply(haps, function(x) x$hap1)
  sv_data[[paste0(sample, "_hap2")]] <- sapply(haps, function(x) x$hap2)
  sv_data[[paste0(sample, "_carrier")]] <- 
    sv_data[[paste0(sample, "_hap1")]] == 1 | sv_data[[paste0(sample, "_hap2")]] == 1
}
cat("Complete!\n\n")

cat("Classifying SVs...\n")

sv_data <- sv_data %>%
  mutate(
    # Ancestral meltilinage (from LA1/LA2)
    ancestral_meltilinage = (LA1_carrier | LA2_carrier) & 
      (PA1_carrier | GA1_carrier | EA5_carrier | CA1_carrier),
    
    # Introduced via WA1 (G1.5 introduction)
    introduced_via_WA1 = WA1_carrier & !LA1_carrier & !LA2_carrier &
      (PA1_carrier | GA1_carrier | EA5_carrier | CA1_carrier),
    
    # Introduced via EA3 (G2 introduction)
    introduced_via_EA3 = EA3_carrier & GA1_carrier & 
      !LA1_carrier & !LA2_carrier & !PA1_carrier & !WA1_carrier &
      (EA5_carrier | CA1_carrier),
    
    # Introduced via WA2 (G4 introduction)
    introduced_via_WA2 = WA2_carrier & CA1_carrier & 
      !LA1_carrier & !LA2_carrier & !PA1_carrier & !WA1_carrier &
      !GA1_carrier & !EA5_carrier,
    
    # De novo events by generation
    de_novo_G5 = CA1_carrier & !EA5_carrier & !WA2_carrier &
      !GA1_carrier & !PA1_carrier & !WA1_carrier &
      !LA1_carrier & !LA2_carrier,
    
    de_novo_G4 = EA5_carrier & !GA1_carrier & !PA1_carrier & !WA1_carrier &
      !LA1_carrier & !LA2_carrier & !CA1_carrier,
    
    de_novo_G3 = GA1_carrier & !PA1_carrier & !WA1_carrier & !EA3_carrier &
      !LA1_carrier & !LA2_carrier,
    
    de_novo_G2 = PA1_carrier & !WA1_carrier & !LA1_carrier & !LA2_carrier,
    
    de_novo_WA1 = WA1_carrier & !LA1_carrier & !LA2_carrier &
      !PA1_carrier & !GA1_carrier & !EA5_carrier & !CA1_carrier,
    
    # Lost in meltilinage
    lost_in_meltilinage = (LA1_carrier | LA2_carrier) & 
      !WA1_carrier & !PA1_carrier & !GA1_carrier & !EA5_carrier & !CA1_carrier,
    
    # Private variants
    private_LA1 = LA1_carrier & !LA2_carrier & !PA1_carrier & !GA1_carrier & !EA5_carrier & !CA1_carrier,
    private_LA2 = LA2_carrier & !LA1_carrier & !PA1_carrier & !GA1_carrier & !EA5_carrier & !CA1_carrier,
    private_WA2 = WA2_carrier & !CA1_carrier & !EA5_carrier & !GA1_carrier & 
      !LA1_carrier & !LA2_carrier & !EA3_carrier,
    private_EA3 = EA3_carrier & !GA1_carrier & !EA5_carrier & !CA1_carrier & 
      !LA1_carrier & !LA2_carrier & !WA2_carrier
  )

# Comprehensive classification
sv_data <- sv_data %>%
  mutate(
    sv_class = case_when(
      de_novo_G5 ~ "De novo G5 (CA1)",
      de_novo_G4 ~ "De novo G4 (EA5)",
      de_novo_G3 ~ "De novo G3 (GA1)",
      de_novo_G2 ~ "De novo G2 (PA1)",
      de_novo_WA1 ~ "De novo WA1",
      introduced_via_WA1 ~ "Introduced via WA1 (G1.5)",
      introduced_via_EA3 ~ "Introduced via EA3 (G2)",
      introduced_via_WA2 ~ "Introduced via WA2 (G4)",
      ancestral_meltilinage ~ "Ancestral (Meltilinage)",
      lost_in_meltilinage ~ "Lost in Meltilinage",
      private_LA1 ~ "Private LA1",
      private_LA2 ~ "Private LA2",
      private_WA2 ~ "Private WA2",
      private_EA3 ~ "Private EA3",
      TRUE ~ "Other/Complex"
    ),
    
    sv_type = TYPE,
    sv_size = abs(LEN),
    CHR = factor(CHR, levels = paste0("Pp0", 1:8)),
    
    size_category = case_when(
      sv_size < 100 ~ "Small (<100bp)",
      sv_size < 1000 ~ "Medium (100-1000bp)",
      sv_size < 10000 ~ "Large (1-10kb)",
      TRUE ~ "Very large (>10kb)"
    ),
    
    size_bin = cut(sv_size, breaks = c(0, 100, 500, 1000, 5000, 10000, 50000, Inf),
                   labels = c("<100", "100-500", "500-1000", "1k-5k", "5k-10k", "10k-50k", ">50k"))
  )
